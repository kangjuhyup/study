QUERY PLAN
QueryPlan {
  Sequence {
    Fetch(service: "products") {
      {
        products {
          __typename
          id
          name
          price
          weight
        }
      }
    },
    Flatten(path: "products.@") {
      Fetch(service: "inventory") {
        {
          ... on Product {
            __typename
            id
            price
            weight
          }
        } =>
        {
          ... on Product {
            inStock
            shippingEstimate
          }
        }
      },
    },
  },
}
