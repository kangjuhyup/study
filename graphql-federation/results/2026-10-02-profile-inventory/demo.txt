CLIENT QUERY
query MyInventory {
  me {
    profile { displayName email }
    inventory {
      totalCount
      products { id name price }
    }
  }
}

RESPONSE
{
  "data": {
    "me": {
      "profile": {
        "displayName": "학습 사용자",
        "email": "learner@example.test"
      },
      "inventory": {
        "totalCount": 2,
        "products": [
          {
            "id": "p1",
            "name": "키보드",
            "price": 80000
          },
          {
            "id": "p2",
            "name": "모니터",
            "price": 250000
          }
        ]
      }
    }
  }
}

QUERY PLAN
QueryPlan {
  Sequence {
    Fetch(service: "profile") {
      {
        me {
          __typename
          id
          profile {
            displayName
            email
          }
        }
      }
    },
    Flatten(path: "me") {
      Fetch(service: "inventory") {
        {
          ... on User {
            __typename
            id
          }
        } =>
        {
          ... on User {
            inventory {
              totalCount
              products {
                __typename
                id
              }
            }
          }
        }
      },
    },
    Flatten(path: "me.inventory.products.@") {
      Fetch(service: "products") {
        {
          ... on Product {
            __typename
            id
          }
        } =>
        {
          ... on Product {
            name
            price
          }
        }
      },
    },
  },
}
